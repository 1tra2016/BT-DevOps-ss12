server {
    listen 80;
    server_name api.example.com;

    location /api/export/ {
        proxy_pass http://localhost:8080;

        proxy_connect_timeout 120s;
        proxy_send_timeout 120s;
        proxy_read_timeout 120s;
    }
}

proxy_connect_timeout 120s: thời gian tối đa Nginx chờ kết nối tới Backend.
proxy_send_timeout 120s: thời gian tối đa Nginx chờ khi gửi request tới Backend.
proxy_read_timeout 120s: thời gian tối đa Nginx chờ Backend trả dữ liệu.
